+local function onInteractiveMessage(id, message, buttons)

+        return
+    end
+    waitingMessageId = id

+    local function answer(button)

+            error('Invalid button index: ' .. tostring(button))
+        end
+        waitingMessageId = nil

+        updateInputCapture()
+    end
