# Typing the character's name on the on-screen keyboard in controller menus mode.
launch --start "Balmora, Caius Cosades' House" --set "GUI.controller menus=true"
console command=EnableNameMenu
wait frames=10
wait-mode ChargenName
shot keyboard
expect-mode ChargenName
# The edit box behind the on-screen keyboard has the key focus, so a real keyboard types into it too.
expect-focus TextEdit
pad button=dpright
pad button=dpright
pad button=dpright
pad button=a
pad button=dpdown
pad button=dpleft
pad button=a
pad button=dpdown
pad button=dpleft
pad button=a
wait frames=5
wait-mode ChargenName
shot keyboard-typed
expect-mode ChargenName
