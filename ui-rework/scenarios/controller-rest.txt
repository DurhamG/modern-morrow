# Resting and the level up that follows it, in controller menus mode. The player is hurt and has enough skill
# increases for a level, so resting offers "until healed", and waking up opens the level up.
# The rest happens in a cave, where sleep is allowed and no creature can interrupt it, as one can outdoors. Its
# bandits are disabled, since nobody can rest with enemies nearby. Tribunal's script that sends a Dark Brotherhood
# assassin to attack a sleeping player, at random, is stopped.
launch --start "Addamasartus" --set "GUI.controller menus=true"
lua "code=local types = require('openmw.types'); local cell = require('openmw.world').players[1].cell; for _, kind in ipairs({ types.NPC, types.Creature }) do for _, actor in ipairs(cell:getAll(kind)) do actor.enabled = false end end" context=global
console "command=stopscript dbattackScript"
wait frames=30
console "command=player->modcurrenthealth -30"
lua "code=local types = require('openmw.types') local level = types.Actor.stats.level(require('openmw.self')) level.progress = 10 level.skillIncreasesForAttribute.strength = 6 level.skillIncreasesForAttribute.endurance = 9 level.skillIncreasesForAttribute.agility = 2"
key key=T
wait frames=10
wait-mode Rest
shot rest
expect-mode Rest
pad button=dpright
pad button=dpright
pad button=dpright
wait frames=5
wait-mode Rest
shot rest-more-hours
expect-mode Rest
pad button=rightshoulder
pad button=a
wait frames=10
shot resting
wait frames=300
wait-mode LevelUp timeout=20
shot level-up
expect-mode LevelUp
pad button=a
pad button=dpright
pad button=dpright
pad button=dpright
pad button=a
pad button=dpdown
pad button=a
wait frames=5
wait-mode LevelUp timeout=20
shot level-up-chosen
expect-mode LevelUp
pad button=x
wait frames=20
wait-no-mode
shot level-up-done
expect-no-mode
