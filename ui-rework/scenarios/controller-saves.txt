# Saving and loading from the pause menu in controller menus mode. The scenario saves a new game, looks at it in the
# load list, then starts to save over it, which asks for confirmation.
launch --start "Balmora, Caius Cosades' House" --set "GUI.controller menus=true"
pad button=start
wait frames=10
pad button=a
wait frames=10
shot save
pad button=a
wait frames=60
pad button=start
wait frames=10
pad button=dpdown
pad button=a
wait frames=10
shot load
pad button=b
wait frames=5
pad button=dpup
pad button=a
wait frames=10
pad button=dpdown
wait frames=5
shot save-over
pad button=a
wait frames=10
shot save-over-confirm
pad button=b
wait frames=10
shot save-over-cancelled
