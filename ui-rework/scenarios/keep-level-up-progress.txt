# With the "keep level up progress" setting, a level up spends only the skill increases that the raised attributes'
# multipliers need. Strength has 14 increases, which give x5 and keep 4. Endurance has 6, which give x3 and keep 1.
# Agility has 2 and isn't raised, so it keeps both. The picks on the screen are Strength, Speed and Endurance.
# Without the setting, every attribute starts again from none. The rest happens in a cave as in controller-rest, with
# its bandits disabled and Tribunal's assassin script stopped.
launch --start "Addamasartus" --set "GUI.controller menus=true" --set "Game.keep level up progress=true"
lua "code=local types = require('openmw.types'); local cell = require('openmw.world').players[1].cell; for _, kind in ipairs({ types.NPC, types.Creature }) do for _, actor in ipairs(cell:getAll(kind)) do actor.enabled = false end end" context=global
console "command=stopscript dbattackScript"
wait frames=30
lua "code=local level = require('openmw.types').Actor.stats.level(require('openmw.self')) level.progress = 10 level.skillIncreasesForAttribute.strength = 14 level.skillIncreasesForAttribute.endurance = 6 level.skillIncreasesForAttribute.agility = 2"
key key=T
wait frames=10
wait-mode Rest
pad button=rightshoulder
pad button=a
wait frames=300
wait-mode LevelUp timeout=20
shot level-up
expect-mode LevelUp
pad button=a
pad button=dpright
pad button=dpright
pad button=dpright
pad button=a
pad button=dpdown
pad button=a
wait frames=5
wait-mode LevelUp timeout=20
shot level-up-chosen
pad button=x
wait frames=20
wait-no-mode
expect-lua "local l = require('openmw.types').Actor.stats.level(require('openmw.self')).skillIncreasesForAttribute; print(l.strength .. ' ' .. l.endurance .. ' ' .. l.agility .. ' ' .. l.speed)" "4 1 2 0" context=player
key key=J
wait frames=20
wait-mode Interface
pad axis=lefttrigger value=1 hold=4
wait frames=30
shot kept-progress
