# Turning the "retroactive health" setting on and off in the Options menu, and what each change does to health. The
# player is a Dark Elf Acrobat with Strength 50, Endurance 60 and level 10, who started with 35 health. The setting
# gives (40 + 50) / 2 + 9 * 6 = 99, and current health changes by as much as base health does. Turning it off gives
# back the health Morrowind's rule gives. A werewolf's extra health stays while the setting is on, and a level up
# while it is on counts toward the health Morrowind's rule gives.
launch --start "Addamasartus" --set "GUI.controller menus=true"
lua "code=local types = require('openmw.types'); local cell = require('openmw.world').players[1].cell; for _, kind in ipairs({ types.NPC, types.Creature }) do for _, actor in ipairs(cell:getAll(kind)) do actor.enabled = false end end" context=global
console "command=stopscript dbattackScript"
console "command=player->setstrength 50"
console "command=player->setendurance 60"
console "command=player->setlevel 10"
console "command=player->modcurrenthealth -10"
wait frames=5
expect-lua "local h = require('openmw.types').Actor.stats.dynamic.health(require('openmw.self')); print(string.format('%.1f %.1f', h.base, h.current))" "35.0 25.0" context=player
# Turn the setting on.
pad button=start
wait-mode MainMenu

expect-lua "local h = require('openmw.types').Actor.stats.dynamic.health(require('openmw.self')); print(string.format('%.1f %.1f', h.base, h.current))" "99.0 89.0" context=player
# A werewolf has twice the health, and loses it when they change back.
console "command=player->becomewerewolf"
wait frames=10
expect-lua "local h = require('openmw.types').Actor.stats.dynamic.health(require('openmw.self')); print(string.format('%.1f %.1f', h.base, h.current))" "198.0 198.0" context=player
console "command=player->undowerewolf"
wait frames=10
expect-lua "local h = require('openmw.types').Actor.stats.dynamic.health(require('openmw.self')); print(string.format('%.1f %.1f', h.base, h.current))" "99.0 99.0" context=player
# Turning the setting off takes away the 64 it gave, but leaves at least 1 health.
console "command=player->modcurrenthealth -95"
